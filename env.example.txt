# Firebase Configuration
# Get these values from Firebase Console: https://console.firebase.google.com/
# Project Settings > General > Your apps > Web app

NEXT_PUBLIC_FIREBASE_API_KEY=your_api_key_here
NEXT_PUBLIC_FIREBASE_AUTH_DOMAIN=your_project.firebaseapp.com
NEXT_PUBLIC_FIREBASE_PROJECT_ID=your_project_id
NEXT_PUBLIC_FIREBASE_STORAGE_BUCKET=your_project.appspot.com
NEXT_PUBLIC_FIREBASE_MESSAGING_SENDER_ID=your_sender_id
NEXT_PUBLIC_FIREBASE_APP_ID=your_app_id

# Instructions:
# 1. Go to https://console.firebase.google.com/
# 2. Create a new project (or select existing)
# 3. Go to Project Settings > General
# 4. Scroll down to "Your apps" and click "Web" icon (</>)
# 5. Register your app and copy the config values
# 6. Create a file named .env.local (without .example) in the root directory
# 7. Paste these values with your actual Firebase credentials
